# 1. 基础镜像：使用官方的 Python 3.9  slim 版本（体积小，适合生产环境）
FROM python:3.9-slim

# 2. 设置工作目录：容器里的 /app 目录
WORKDIR /app

# 3. 复制依赖文件：先把 requirements.txt 拷进去
COPY requirements.txt .

# 4. 安装依赖：使用国内源加速下载，防止超时
RUN pip install --no-cache-dir -r requirements.txt -i https://pypi.tuna.tsinghua.edu.cn/simple

# 5. 复制项目代码：把当前目录所有文件拷进容器
COPY . .

# 6. 暴露端口：告诉外界这个容器监听 5000 端口
EXPOSE 5000

# 7. 启动命令：容器启动时自动运行的命令
# 假设你的主程序叫 app.py，如果是 main.py 请自行修改
CMD ["python", "app.py"]
